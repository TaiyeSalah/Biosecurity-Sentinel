# ═══════════════════════════════════════════════════════════════════════════
# Biosecurity Sentinel — Dockerfile
# Multi-stage build: builder installs heavy deps, runtime is lean
# ═══════════════════════════════════════════════════════════════════════════

# ── Stage 1: Builder ─────────────────────────────────────────────────────────
FROM python:3.11-slim AS builder

WORKDIR /build

# System deps needed for RDKit and torch-geometric compilation
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    wget \
    curl \
    git \
    libxrender1 \
    libxext6 \
    && rm -rf /var/lib/apt/lists/*

# Install torch first (large — cache separately)
RUN pip install --no-cache-dir torch==2.3.0 --index-url https://download.pytorch.org/whl/cpu

# Install torch-geometric CPU wheels
RUN pip install --no-cache-dir \
    torch-scatter \
    torch-sparse \
    -f https://data.pyg.org/whl/torch-2.3.0+cpu.html

# Copy and install remaining requirements
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# ── Stage 2: Runtime ─────────────────────────────────────────────────────────
FROM python:3.11-slim AS runtime

WORKDIR /app

# Runtime system deps only
RUN apt-get update && apt-get install -y --no-install-recommends \
    libxrender1 \
    libxext6 \
    && rm -rf /var/lib/apt/lists/*

# Copy installed packages from builder
COPY --from=builder /usr/local/lib/python3.11/site-packages /usr/local/lib/python3.11/site-packages
COPY --from=builder /usr/local/bin /usr/local/bin

# Copy application source
COPY api/        ./api/
COPY core/       ./core/
COPY models/     ./models/
COPY config/     ./config/
COPY training/   ./training/

# Create directories for runtime artifacts
RUN mkdir -p data/ results/ logs/

# Non-root user for security
RUN useradd -m -u 1000 sentinel
RUN chown -R sentinel:sentinel /app
USER sentinel

EXPOSE 8000

# Health check
HEALTHCHECK --interval=30s --timeout=10s --start-period=15s --retries=3 \
    CMD curl -f http://localhost:8000/health || exit 1

CMD ["uvicorn", "api.main:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "1"]
